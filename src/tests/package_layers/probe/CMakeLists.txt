cmake_minimum_required(VERSION 3.24 FATAL_ERROR)

### Probe of the package-layer tests: configured against the build tree of the
### stub project in .. and checks what its OpenMSConfig.cmake reports.
project("OpenMS_PackageLayersProbe" CXX)

option(EXPECT_CLI "The build tree was configured with WITH_CLI=ON, so the CLI layer has to be there" ON)
option(EXPECT_GUI "The build tree has the stub OpenMSGUI package, so the GUI component has to be found" OFF)
option(REQUIRE_CLI "Require the CLI component, which a WITH_CLI=OFF build tree has to refuse" OFF)
option(REQUIRE_GUI "Require the GUI component, which a build tree without the GUI package has to refuse" OFF)
option(NO_COMPONENTS "Do not request any component: the GUI layer is then provided without its package configuration" OFF)

## each has to fail: the driving test passes on the reason OpenMSConfig.cmake gives
if(REQUIRE_CLI)
  find_package(OpenMS CONFIG REQUIRED COMPONENTS CLI)
  message(FATAL_ERROR "find_package(OpenMS COMPONENTS CLI) succeeded on a build tree configured with WITH_CLI=OFF")
endif()
if(REQUIRE_GUI)
  find_package(OpenMS CONFIG REQUIRED COMPONENTS GUI)
  message(FATAL_ERROR "find_package(OpenMS COMPONENTS GUI) succeeded on a build tree without the GUI package")
endif()

if(NO_COMPONENTS)
  find_package(OpenMS CONFIG REQUIRED)
else()
  find_package(OpenMS CONFIG REQUIRED OPTIONAL_COMPONENTS CLI GUI)
endif()

foreach(_target IN ITEMS OpenMS::OpenMS OpenMS::OpenSwathAlgo)
  if(NOT TARGET ${_target})
    message(FATAL_ERROR "the build tree's package did not provide the target ${_target}")
  endif()
endforeach()
foreach(_var IN ITEMS OpenMS_VERSION_FULLSTRING OpenMS_VERSION_PRERELEASE)
  if(NOT DEFINED ${_var})
    message(FATAL_ERROR "the build tree's package does not set ${_var}")
  endif()
endforeach()

if(EXPECT_CLI)
  if(NOT OpenMS_CLI_FOUND OR NOT TARGET OpenMS::OpenMS_CLI)
    message(FATAL_ERROR "the build tree was configured with WITH_CLI=ON but its package does not provide the CLI layer")
  endif()
else()
  if(OpenMS_CLI_FOUND)
    message(FATAL_ERROR "the build tree was configured with WITH_CLI=OFF but its package reports OpenMS_CLI_FOUND=${OpenMS_CLI_FOUND}")
  endif()
  foreach(_target IN ITEMS OpenMS::OpenMS_CLI OpenMS_CLI)
    if(TARGET ${_target})
      message(FATAL_ERROR "the build tree was configured with WITH_CLI=OFF but its package provides the target ${_target} (stale CLI target file)")
    endif()
  endforeach()
endif()

if(EXPECT_GUI AND NO_COMPONENTS)
  if(NOT OpenMS_WITH_GUI OR NOT TARGET OpenMS::OpenMS_GUI OR NOT TARGET OpenMS_GUI)
    message(FATAL_ERROR "the build tree has the OpenMSGUI package but its targets were not provided without the component GUI")
  endif()
  if(OpenMS_GUI_FOUND)
    message(FATAL_ERROR "the component GUI was not requested but OpenMS_GUI_FOUND=${OpenMS_GUI_FOUND}")
  endif()
  if(OpenMSGUI_STUB_CONFIG_LOADED)
    message(FATAL_ERROR "the component GUI was not requested but the OpenMSGUI package configuration (with its Qt lookup) was loaded")
  endif()
elseif(EXPECT_GUI)
  if(NOT OpenMS_GUI_FOUND OR NOT OpenMS_WITH_GUI OR NOT TARGET OpenMS::OpenMS_GUI OR NOT TARGET OpenMS_GUI)
    message(FATAL_ERROR "the build tree has the OpenMSGUI package but the GUI component was not found")
  endif()
else()
  foreach(_var IN ITEMS OpenMS_WITH_GUI OpenMS_GUI_FOUND)
    if(${_var})
      message(FATAL_ERROR "the build tree has no OpenMSGUI package but its package reports ${_var}=${${_var}}")
    endif()
  endforeach()
  foreach(_target IN ITEMS OpenMS::OpenMS_GUI OpenMS_GUI)
    if(TARGET ${_target})
      message(FATAL_ERROR "the build tree has no OpenMSGUI package but provides the target ${_target}")
    endif()
  endforeach()
endif()
message(STATUS "package layers as expected (CLI: ${OpenMS_CLI_FOUND}, GUI: ${OpenMS_GUI_FOUND}, GUI layer: ${OpenMS_WITH_GUI})")
